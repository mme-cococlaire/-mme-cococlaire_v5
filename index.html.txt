function showTheme(themeKey){
var lesson=lessons[themeKey];
if(!lesson)return;
document.getElementById('home-page').style.display='none';
document.getElementById('lesson-page').classList.add('active');
document.getElementById('lesson-body').innerHTML=lesson.content;
window.scrollTo(0,0);
}

function showNiveau(niveauKey){
var niveau=niveaux[niveauKey];
if(!niveau)return;
var lessonsList='';
for(var i=0;i<niveau.lessons.length;i++){
var key=niveau.lessons[i];
var l=lessons[key];
lessonsList+='<div class="theme-card" onclick="showTheme(''+key+'')" style="cursor:pointer;background:#f0f0ff;padding:15px;border-radius:10px;margin:10px 0;"><h3>'+l.title+'</h3></div>';
}
document.getElementById('home-page').style.display='none';
document.getElementById('lesson-page').classList.add('active');
document.getElementById('lesson-body').innerHTML='<button class="back-btn" onclick="goHome()">Retour</button><h2 style="color:#764ba2;margin-bottom:20px;">'+niveau.title+' - Lecons disponibles</h2><div>'+lessonsList+'</div>';
window.scrollTo(0,0);
}

function goHome(){
document.getElementById('home-page').style.display='block';
document.getElementById('lesson-page').classList.remove('active');
window.scrollTo(0,0);
}

function checkAnswer(element,isCorrect){
var quizSection=element.closest('.quiz-section');
var options=quizSection.querySelectorAll('.quiz-option');
for(var i=0;i<options.length;i++){
options[i].classList.remove('correct','wrong');
options[i].style.pointerEvents='none';
}
if(isCorrect){
element.classList.add('correct');
element.innerHTML+=' OK';
}else{
element.classList.add('wrong');
element.innerHTML+=' X';
for(var j=0;j<options.length;j++){
var onclickStr=options[j].getAttribute('onclick')||'';
if(onclickStr.indexOf('true')!==-1){
options[j].classList.add('correct');
}
}
}
}

function addMessage(text,isUser){
var messagesDiv=document.getElementById('chatMessages');
var msgDiv=document.createElement('div');
msgDiv.className='message'+(isUser?' user':'');
msgDiv.innerHTML='<div class="bubble">'+text+'</div>';
messagesDiv.appendChild(msgDiv);
messagesDiv.scrollTop=messagesDiv.scrollHeight;
}

function sendMessage(){
var input=document.getElementById('chatInput');
var text=input.value.trim();
if(!text)return;
addMessage(text,true);
input.value='';
setTimeout(function(){
var response=getBotResponse(text);
addMessage(response,false);
},500);
}

function sendSuggestion(text){
addMessage(text,true);
setTimeout(function(){
var response=getBotResponse(text);
addMessage(response,false);
},500);
}

function getBotResponse(input){
var lower=input.toLowerCase();
if(lower.indexOf('nom')!==-1||lower.indexOf('genre')!==-1){
return 'Un nom designe une personne, un animal, une chose ou une idee. Il a un genre (masculin/feminin) et un nombre (singulier/pluriel).';
}
if(lower.indexOf('passe compose')!==-1||lower.indexOf('passe compos')!==-1){
return 'Le passe compose se forme avec AVOIR ou ETRE + participe passe. Ex: "J ai mange", "Je suis alle(e)".';
}
if(lower.indexOf('c est')!==-1||lower.indexOf('s est')!==-1){
return '"C est" = ce + est (demonstratif + etre). "S est" = se + est (reflechi + etre).';

return 'Figures de style : Comparaison (comme), Metaphore (sans mot de liaison), Personnification (donner des qualites humaines), Hyperbole (exageration).';
}

return 'L adjectif s accorde en genre et en nombre avec le nom. Ex: noir -> noire -> noirs -> noires.';
}

}

}
